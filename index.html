-- Shoot Board v2: sign-in, invite-only team list, activity log -- Paste all of this into Supabase → SQL Editor → New query, then click Run. -- NOTE: this resets the board (old test people and events are removed). drop table if exists public.activity cascade; drop table if exists public.events cascade; drop table if exists public.people cascade; drop table if exists public.members cascade; -- 1. The team list: only these emails can use the board ------------------ create table public.members ( email text primary key, is_admin boolean not null default false, added_at timestamptz not null default now() ); -- You, as organiser. Change this if you sign in with a different email. insert into public.members (email, is_admin) values ('jacoborahilly@gmail.com', true); create or replace function public.my_email() returns text language sql stable as $$ select lower(coalesce(auth.jwt() ->> 'email', '')) $$; create or replace function public.is_member() returns boolean language sql stable security definer set search_path = public as $$ select exists (select 1 from members where lower(email) = my_email()) $$; create or replace function public.is_admin() returns boolean language sql stable security definer set search_path = public as $$ select exists (select 1 from members where lower(email) = my_email() and is_admin) $$; -- 2. People: one row per signed-in person -------------------------------- create table public.people ( user_id uuid primary key default auth.uid() references auth.users(id) on delete cascade, email text not null default public.my_email(), name text not null default '', role text, slots text[] not null default '{}', updated_at timestamptz not null default now() ); -- 3. Events --------------------------------------------------------------- create table public.events ( id uuid primary key default gen_random_uuid(), title text not null, type text not null default 'Shoot', date date not null, call text, wrap text, loc text, roles text[] not null default '{}', notes text, created_by uuid default auth.uid() references auth.users(id) on delete set null, created_at timestamptz not null default now() ); -- 4. Activity log (written automatically, nobody can edit it) ------------- create table public.activity ( id bigint generated always as identity primary key, at timestamptz not null default now(), user_id uuid, actor text not null, action text not null, detail text ); create or replace function public.log_activity() returns trigger language plpgsql security definer set search_path = public as $$ declare nm text; act text; det text; recent bigint; begin select nullif(name, '') into nm from people where user_id = auth.uid(); nm := coalesce(nm, nullif(my_email(), ''), 'Someone'); if tg_table_name = 'events' then if tg_op = 'INSERT' then act := 'added'; det := new.type || ': ' || new.title || ' (' || new.date::text || ')'; elsif tg_op = 'UPDATE' then act := 'edited'; det := new.type || ': ' || new.title; else act := 'removed'; det := old.type || ': ' || old.title || ' (' || old.date::text || ')'; end if; else if tg_op = 'INSERT' then act := 'joined the board'; det := null; elsif tg_op = 'UPDATE' then if new.slots is distinct from old.slots then -- fold rapid taps into one entry select id into recent from activity where user_id = auth.uid() and action = 'updated their availability' and at > now() - interval '15 minutes' order by at desc limit 1; if recent is not null then update activity set at = now(), actor = nm where id = recent; return null; end if; act := 'updated their availability'; det := null; elsif new.name is distinct from old.name or new.role is distinct from old.role then act := 'updated their profile'; det := null; else return null; end if; else act := 'removed crew member'; det := coalesce(nullif(old.name, ''), old.email); end if; end if; insert into activity (user_id, actor, action, detail) values (auth.uid(), nm, act, det); return null; end $$; create trigger events_log after insert or update or delete on public.events for each row execute function public.log_activity(); create trigger people_log after insert or update or delete on public.people for each row execute function public.log_activity(); -- 5. Access rules: signed in AND on the team list --------------------------- alter table public.members enable row level security; alter table public.people enable row level security; alter table public.events enable row level security; alter table public.activity enable row level security; create policy "see own membership, admins see all" on public.members for select to authenticated using (lower(email) = my_email() or is_admin()); create policy "admins add members" on public.members for insert to authenticated with check (is_admin()); create policy "admins remove members" on public.members for delete to authenticated using (is_admin()); create policy "team reads people" on public.people for select to authenticated using (is_member()); create policy "add yourself" on public.people for insert to authenticated with check (is_member() and user_id = auth.uid()); create policy "edit yourself" on public.people for update to authenticated using (user_id = auth.uid() and is_member()); create policy "remove self or admin" on public.people for delete to authenticated using (user_id = auth.uid() or is_admin()); create policy "team reads events" on public.events for select to authenticated using (is_member()); create policy "team adds events" on public.events for insert to authenticated with check (is_member() and created_by = auth.uid()); create policy "creator or admin edits" on public.events for update to authenticated using (created_by = auth.uid() or is_admin()); create policy "creator or admin removes" on public.events for delete to authenticated using (created_by = auth.uid() or is_admin()); create policy "team reads activity" on public.activity for select to authenticated using (is_member()); -- 6.
